# ---------------------------------------------------------------------------
# syntaxError: compiler built-in function-like macros cppcheck cannot evaluate
# ---------------------------------------------------------------------------
# memory_macros.h uses __has_cpp_attribute(...) and __has_attribute(...) in
# #if conditions.  These are Clang/GCC/MSVC built-ins, not ordinary macros;
# cppcheck 2.x cannot expand them in preprocessor conditionals and reports a
# syntax error.  The guarded paths (C++20 [[likely]]/[[unlikely]] attributes
# and thread-safety annotations) are correct; suppress the false parse failure.
syntaxError:*memory_macros.h

# memory_allocator.cpp uses the MEMORY_UNLIKELY macro as `if MEMORY_UNLIKELY
# (cond)` — valid after expansion to `if (__builtin_expect(...))` or the C++20
# attribute form, but cppcheck sees the unexpanded token and rejects it.
syntaxError:*memory_allocator.cpp

# gpu/cuda_caching_allocator.cpp has the same MEMORY_UNLIKELY pattern.
syntaxError:*cuda_caching_allocator.cpp

# copy_token.h triggers a spurious "namespacememory{" parse error only when
# certain include combinations are analysed together — cppcheck token-merging
# artifact, not a real issue.
syntaxError:*copy_token.h

normalCheckLevelMaxBranches

# ---------------------------------------------------------------------------
# unusedFunction: library API functions
# ---------------------------------------------------------------------------
# All public API functions (allocator helpers, copy_sync/copy_async, workspace
# accessors, caching-allocator stat helpers, NUMA utilities, etc.) are only
# called by consumers outside the include/ tree.  cppcheck analyses include/
# in isolation and cannot see those callers, so every public entry-point
# appears "unused".  Suppress globally rather than annotating every header.
unusedFunction

# ---------------------------------------------------------------------------
# style: intentional design choices
# ---------------------------------------------------------------------------
# functionStatic: cppcheck often flags member functions that could be static,
# but this ignores namespace/friend scoping, and converting methods to static
# can break encapsulation. Suppress globally.
functionStatic

# useStlAlgorithm: simple hand-written loops are often clearer and more
# efficient than STL algorithms, especially in hot paths. We prefer
# readability and measured performance over algorithmic style.
useStlAlgorithm

# shadowFunction: local variables shadowing outer names is a common pattern
# and usually intentional (e.g., reusing names in inner scopes).
shadowFunction

# variableScope: reducing variable scope is a style preference; cppcheck
# is too aggressive here and we prefer declaration at the point of first use.
variableScope

# constVariablePointer: in cuda_caching_allocator.cpp trim_cache_locked() and
# release_all_blocks(), the loop variable 'pool' iterates over pointers to
# mutable pools (small_blocks_, large_blocks_). Those walks only read the
# pools or clear them at teardown; the pool objects stay non-const because
# other paths erase through block->pool (victim->pool->blocks.erase()).
constVariablePointer:src/gpu/cuda_caching_allocator.cpp:2002
constVariablePointer:src/gpu/cuda_caching_allocator.cpp:2059

# signal ownership transfer, consistent with platform allocator APIs. Line
# numbers shift as the file changes; re-point here rather than deleting on drift.
constParameterPointer:src/helper/pinned_memory_allocator.cpp:385
constParameterPointer:src/helper/pinned_memory_allocator.cpp:523

# constParameterPointer: transfer.cpp's validate_stream_handle() takes

constParameterPointer:src/transfer.cpp:163

# knownEmptyContainer: in cuda_caching_allocator.cpp:1679 and 1693, cppcheck

# knownConditionTrueFalse: profiling_gate.h's countdown() returns a
# thread_local static that persists and is decremented across calls; cppcheck
# analyses the function body alone and treats the fresh-looking static as
# always its initializer value. False positive — the whole point of the gate
# is that this condition is true on most calls.
knownConditionTrueFalse:include/profiler/profiling_gate.h

# knownConditionTrueFalse: cuda_caching_allocator.cpp's pending_event is
# reassigned on every lambda iteration (acquire → record → push_back →
# nullptr) and can be non-null when cudaEventRecord/push_back throws
# mid-iteration; cppcheck's per-lambda analysis does not track mutation of the
# outer captured variable across a throwing call inside it. False positive.
knownConditionTrueFalse:src/gpu/cuda_caching_allocator.cpp
